Request: Write isLeapYear(year) for the Gregorian calendar. Positive integer years divisible by 400, or by 4 but not 100, are leap years; invalid years return false.

function isLeapYear(year) {
  return Number.isInteger(year) && year > 0 && (year % 400 === 0 || (year % 4 === 0 && year % 100 !== 0));
}
