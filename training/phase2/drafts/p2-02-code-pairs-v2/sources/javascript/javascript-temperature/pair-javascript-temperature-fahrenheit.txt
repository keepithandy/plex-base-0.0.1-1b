Request: Write toFahrenheit(celsius), converting a finite Celsius number to Fahrenheit. Return null for non-number or non-finite input.

function toFahrenheit(celsius) {
  if (!Number.isFinite(celsius)) return null;
  return celsius * 9 / 5 + 32;
}
